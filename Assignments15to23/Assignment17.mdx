WITH
MEMBER [Measures].[Streams YoY %] AS
  IIF(
      IsEmpty([Measures].[PreviousStreams]) OR [Measures].[PreviousStreams] = 0,
      NULL,
      ([Measures].[Streams First Month] - [Measures].[PreviousStreams])
      / [Measures].[PreviousStreams]
  ),
  FORMAT_STRING = "Percent"

SELECT
{
  [Measures].[Streams First Month],
  [Measures].[PreviousStreams],
  [Measures].[Streams YoY %]
} ON COLUMNS,

NON EMPTY
(
  ([Published Song Fact].[Category].Members - { [Published Song Fact].[Category].[All] })
  *
  ([Dim Date].[Year].Members - { [Dim Date].[Year].[All] })
) ON ROWS

FROM [Group4_DSV];
