
WITH
/*Seasons e Artists senza All */
SET [SeasonsNoAll] AS
  ( [Dim Date].[Season].Members - { [Dim Date].[Season].[All] } )
  
SET [ArtistsNoAll] AS
  ( [Dim Artist].[Name].Members - { [Dim Artist].[Name].[All] } )

/* Media Streams dell'artista sulle Season */
MEMBER [Measures].[Artist Season Avg] AS
AVG(
  [SeasonsNoAll],
  [Measures].[Streams First Month]
)

/* Delta % vs media dell'artista */
MEMBER [Measures].[Delta vs Artist Avg %] AS
IIF(
  IsEmpty([Measures].[Artist Season Avg]) OR [Measures].[Artist Season Avg] = 0,
  NULL,
  ([Measures].[Streams First Month] - [Measures].[Artist Season Avg])
  / [Measures].[Artist Season Avg]
),
FORMAT_STRING = "Percent"

/* Max Delta % dell'artista */
MEMBER [Measures].[Max Delta Artist %] AS
MAX(
  [SeasonsNoAll],
  [Measures].[Delta vs Artist Avg %]
)

SELECT
{
  [Measures].[Streams First Month],
  [Measures].[Artist Season Avg],
  [Measures].[Delta vs Artist Avg %]
} ON COLUMNS,

NON EMPTY
FILTER
(
  (
    [ArtistsNoAll]
    *
    [SeasonsNoAll]
  ),
  /* teniamo solo la season che è il massimo per l'artista */
  [Measures].[Delta vs Artist Avg %] = [Measures].[Max Delta Artist %]
  AND
  /* e solo se è sopra la  media */
  [Measures].[Delta vs Artist Avg %] > 0
) ON ROWS

FROM [Group4_DSV];
